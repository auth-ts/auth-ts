You need Postgres and an app with a server route that takes a `Request` and
returns a `Response`.

<Steps>
<Step>

### Install

```package-install
@auth-ts/core
```

</Step>
<Step>

### Generate a signing key

```bash
npx @auth-ts/cli keygen
```

Answer yes. It appends `JWT_PRIVATE_KEY` to `.env` and writes the public key
set to `public/jwks.json`, which your app then serves at `/jwks.json`.

</Step>

### Mount the handler

Serve every method under `/api/auth`.

```ts tab="TanStack Start" title="src/routes/api/auth/$.ts"
import { createFileRoute } from "@tanstack/react-router"
import { auth } from "~/lib/auth"

export const Route = createFileRoute("/api/auth/$")({
  server: {
    handlers: {
      ANY: ({ request }) => auth.handler(request)
    }
  }
})
```

```ts tab="Next.js" title="app/api/auth/[...all]/route.ts"
import { auth } from "@/lib/auth"

export const GET = auth.handler
export const POST = auth.handler
export const DELETE = auth.handler
```

```ts tab="SolidStart" title="src/routes/api/auth/[...auth].ts"
import type { APIEvent } from "@solidjs/start/server"
import { auth } from "~/lib/auth"

const handler = ({ request }: APIEvent) => auth.handler(request)

export { handler as DELETE, handler as GET, handler as POST }
```

```ts tab="Hono" title="src/index.ts"
import { Hono } from "hono"
import { auth } from "./lib/auth"

const app = new Hono()

app.on(["GET", "POST", "DELETE"], "/api/auth/*", (c) => auth.handler(c.req.raw))

export default app
```

### Query with the token

Point your database at `https://your.app/jwks.json`. On Neon, that is the Data
API's JWKS URL; see [Neon](/docs/data/neon). Then give a table a policy:

```sql
create table "todos" (
  "id" uuid primary key default gen_random_uuid(),
  "userId" uuid not null default (auth.user_id()::uuid) references "users",
  "title" text not null
);

alter table "todos" enable row level security;

create policy "own todos" on "todos" for all to authenticated
  using ((select auth.user_id()::uuid) = "userId")
  with check ((select auth.user_id()::uuid) = "userId");
```

Your data client sends the token on every request, and gets back only the
signed-in user's rows:

```ts
import { fetchWithToken, NeonPostgrestClient } from "@neondatabase/postgrest-js"

const postgrest = new NeonPostgrestClient({
  dataApiUrl: DATA_API_URL,
  options: { global: { fetch: fetchWithToken(authClient.getToken) } }
})

const { data: todos } = await postgrest.from("todos").select()
```
